11pub(crate) struct TypeWithPunctuatedMeta {
12    pub(crate) ty:   Type,
13    pub(crate) list: Punctuated<Meta, Token![,]>,
14}

16impl Parse for TypeWithPunctuatedMeta {
17    #[inline]
18    fn parse(input: ParseStream) -> syn::Result<Self> {
19        let ty = input.parse::<Type>()?;
20
21        if input.is_empty() {
22            return Ok(Self {
23                ty,
24                list: Punctuated::new(),
25            });
26        }
27
28        input.parse::<Token![,]>()?;
29
30        let list = input.parse_terminated(Meta::parse, Token![,])?;
31
32        Ok(Self {
33            ty,
34            list,
35        })
36    }
37}

39/// Describes, for one trait, the places where the automatic bound engine can treat a field type specially instead of emitting a plain `FieldType: Trait` predicate.
40///
41/// An unconditional field type produces no where predicate at all, because the predicate would always hold and would only be noise in the generated code.
42/// A forwarding field type produces the predicates of its type arguments instead, because the standard library guarantees that the arguments satisfying the trait is enough for the whole type.
43/// The unconditional table is also used when a self-referencing field type is degraded to per-parameter bounds, to avoid collecting parameters from unconditional positions.
44pub(crate) struct BoundExceptions {
45    /// Type names, matched against the last path segment, whose implementation of the trait never depends on their type arguments, e.g. `Arc` for `Clone`.
46    pub(crate) unconditional_types:               &'static [&'static str],
47    /// Type names, matched against the last path segment, that implement the trait whenever their type arguments do, e.g. `Option` for `Copy` or `Vec` for `Clone`.
48    ///
49    /// Bounding the arguments instead of the whole type keeps the where clause simple and lets the compiler use supertrait elaboration on the parameters, e.g. deriving `T: Clone` from `T: Copy`.
50    pub(crate) forwarding_types:                  &'static [&'static str],
51    /// Whether a shared reference (`&T`) implements the trait unconditionally, which is the case for `Copy` and `Clone`.
52    pub(crate) shared_reference_is_unconditional: bool,
53}
54
55impl BoundExceptions {
56    /// Checks whether the last segment of the path names a type that implements the trait unconditionally.
57    ///
58    /// `PhantomData` implements every trait supported by Educe unconditionally, so it is always treated as an exception.
59    #[inline]
60    fn path_is_unconditional(&self, path: &Path) -> bool {
61        if let Some(segment) = path.segments.last() {
62            let ident = &segment.ident;
63
64            if ident == "PhantomData" {
65                return true;
66            }
67
68            self.unconditional_types.iter().any(|name| ident == name)
69        } else {
70            false
71        }
72    }
73
74    /// Checks whether the whole type is known to implement the trait unconditionally, so its predicate can be omitted.
75    ///
76    /// Raw pointers and bare function pointers are unconditional for every trait supported by Educe, because their implementations only look at the address value.
77    #[inline]
78    pub(crate) fn type_is_unconditional(&self, ty: &Type) -> bool {
79        match ty {
80            Type::Path(ty) => ty.qself.is_none() && self.path_is_unconditional(&ty.path),
81            Type::Ptr(_) | Type::BareFn(_) => true,
82            Type::Reference(ty) => {
83                ty.mutability.is_none() && self.shared_reference_is_unconditional
84            },
85            _ => false,
86        }
87    }
88
89    /// Returns the type arguments of a type that forwards the trait to them, or `None` if the type is not in the forwarding table.
90    pub(crate) fn forwarding_type_arguments<'a>(&self, ty: &'a Type) -> Option<Vec<&'a Type>> {
91        if let Type::Path(ty) = ty
92            && ty.qself.is_none()
93            && let Some(segment) = ty.path.segments.last()
94            && self.forwarding_types.iter().any(|name| segment.ident == name)
95            && let PathArguments::AngleBracketed(args) = &segment.arguments
96        {
97            let mut types = Vec::new();
98
99            for arg in &args.args {
100                if let GenericArgument::Type(ty) = arg {
101                    types.push(ty);
102                }
103            }
104
105            return Some(types);
106        }
107
108        None
109    }
110}

112/// Walks a type and collects every ident that could refer to a generic type parameter.
113///
114/// When `exceptions` is provided, positions that the exception table marks as unconditional are not descended into, so parameters that only appear there are not collected.
115fn walk_type<'a>(set: &mut HashSet<&'a Ident>, ty: &'a Type, exceptions: Option<&BoundExceptions>) {
116    match ty {
117        Type::Array(ty) => walk_type(set, ty.elem.as_ref(), exceptions),
118        Type::Group(ty) => walk_type(set, ty.elem.as_ref(), exceptions),
119        Type::Paren(ty) => walk_type(set, ty.elem.as_ref(), exceptions),
120        Type::Slice(ty) => walk_type(set, ty.elem.as_ref(), exceptions),
121        Type::Tuple(ty) => {
122            for ty in &ty.elems {
123                walk_type(set, ty, exceptions);
124            }
125        },
126        Type::Reference(ty) => {
127            // A shared reference implements `Copy`/`Clone` no matter what it points to, so those traits do not need the pointee's parameters.
128            if let Some(exceptions) = exceptions
129                && ty.mutability.is_none()
130                && exceptions.shared_reference_is_unconditional
131            {
132                return;
133            }
134
135            walk_type(set, ty.elem.as_ref(), exceptions);
136        },
137        Type::Ptr(ty) => {
138            // Raw pointer impls only compare or print the address, so no trait supported by Educe needs the pointee's parameters.
139            if exceptions.is_some() {
140                return;
141            }
142
143            walk_type(set, ty.elem.as_ref(), exceptions);
144        },
145        Type::BareFn(ty) => {
146            // Function pointer impls also only look at the address.
147            if exceptions.is_some() {
148                return;
149            }
150
151            for arg in &ty.inputs {
152                walk_type(set, &arg.ty, exceptions);
153            }
154
155            if let ReturnType::Type(_, ty) = &ty.output {
156                walk_type(set, ty, exceptions);
157            }
158        },
159        Type::ImplTrait(ty) => {
160            for b in &ty.bounds {
161                if let TypeParamBound::Trait(b) = b {
162                    walk_path(set, &b.path, exceptions);
163                }
164            }
165        },
166        Type::TraitObject(ty) => {
167            for b in &ty.bounds {
168                if let TypeParamBound::Trait(b) = b {
169                    walk_path(set, &b.path, exceptions);
170                }
171            }
172        },
173        Type::Path(ty) => {
174            if let Some(qself) = &ty.qself {
175                walk_type(set, qself.ty.as_ref(), exceptions);
176            }
177
178            walk_path(set, &ty.path, exceptions);
179        },
180        _ => (),
181    }
182}

184/// Walks a path, collecting the first segment ident (the only position where a generic type parameter can appear) and recursing into every segment's arguments.
185fn walk_path<'a>(
186    set: &mut HashSet<&'a Ident>,
187    path: &'a Path,
188    exceptions: Option<&BoundExceptions>,
189) {
190    if let Some(exceptions) = exceptions
191        && exceptions.path_is_unconditional(path)
192    {
193        return;
194    }
195
196    // A path that starts with `::` can never begin with a generic type parameter.
197    if path.leading_colon.is_none()
198        && let Some(segment) = path.segments.first()
199    {
200        // This may also collect idents like `Vec` or `std`, but the callers intersect the set with the real generic parameters, so false candidates are harmless.
201        set.insert(&segment.ident);
202    }
203
204    for segment in &path.segments {
205        match &segment.arguments {
206            PathArguments::AngleBracketed(args) => {
207                for arg in &args.args {
208                    match arg {
209                        GenericArgument::Type(ty) => walk_type(set, ty, exceptions),
210                        GenericArgument::AssocType(ty) => walk_type(set, &ty.ty, exceptions),
211                        _ => (),
212                    }
213                }
214            },
215            PathArguments::Parenthesized(args) => {
216                for ty in &args.inputs {
217                    walk_type(set, ty, exceptions);
218                }
219
220                if let ReturnType::Type(_, ty) = &args.output {
221                    walk_type(set, ty, exceptions);
222                }
223            },
224            PathArguments::None => (),
225        }
226    }
227}

229/// Collects every ident in the type that could refer to a generic type parameter, without applying any exceptions.
230#[inline]
231pub(crate) fn find_all_idents_in_type<'a>(set: &mut HashSet<&'a Ident>, ty: &'a Type) {
232    walk_type(set, ty, None);
233}

235/// Collects the idents in the type that could refer to a generic type parameter, skipping positions that the exception table marks as unconditional.
236#[inline]
237pub(crate) fn find_idents_in_type<'a>(
238    set: &mut HashSet<&'a Ident>,
239    ty: &'a Type,
240    exceptions: &BoundExceptions,
241) {
242    walk_type(set, ty, Some(exceptions));
243}

245/// Collects only a bare, single-segment type ident such as `T`, without descending into any structure.
246///
247/// This is used by the `Into` handler: a bound like `T: Into<Target>` only makes sense when the field type is the parameter itself, so nested parameters like the `T` in `Option<T>` must not be collected.
248#[inline]
249pub(crate) fn find_bare_ident_in_type<'a>(set: &mut HashSet<&'a Ident>, ty: &'a Type) {
250    if let Type::Path(ty) = ty
251        && ty.qself.is_none()
252        && let Some(ident) = ty.path.get_ident()
253    {
254        set.insert(ident);
255    }
256}

258/// Returns true if the type syntactically uses any of the generic type parameters.
259#[inline]
260pub(crate) fn type_uses_type_params(ty: &Type, params: &Punctuated<GenericParam, Comma>) -> bool {
261    let mut set = HashSet::new();
262
263    find_all_idents_in_type(&mut set, ty);
264
265    params.iter().any(|param| {
266        if let GenericParam::Type(param) = param { set.contains(&param.ident) } else { false }
267    })
268}

270/// Returns true if any path segment in the type is exactly the given ident.
271///
272/// This is used to detect field types that refer to the type currently being derived, e.g. `Box<List<T>>` inside `List<T>`, so that the bound engine can avoid generating a self-referencing predicate that would overflow the trait solver.
273pub(crate) fn type_mentions_ident(ty: &Type, ident: &Ident) -> bool {
274    fn path_mentions_ident(path: &Path, ident: &Ident) -> bool {
275        for segment in &path.segments {
276            if segment.ident == *ident {
277                return true;
278            }
279
280            match &segment.arguments {
281                PathArguments::AngleBracketed(args) => {
282                    for arg in &args.args {
283                        match arg {
284                            GenericArgument::Type(ty) => {
285                                if type_mentions_ident(ty, ident) {
286                                    return true;
287                                }
288                            },
289                            GenericArgument::AssocType(ty)
290                                if type_mentions_ident(&ty.ty, ident) =>
291                            {
292                                return true;
293                            },
294                            _ => (),
295                        }
296                    }
297                },
298                PathArguments::Parenthesized(args) => {
299                    for ty in &args.inputs {
300                        if type_mentions_ident(ty, ident) {
301                            return true;
302                        }
303                    }
304
305                    if let ReturnType::Type(_, ty) = &args.output
306                        && type_mentions_ident(ty, ident)
307                    {
308                        return true;
309                    }
310                },
311                PathArguments::None => (),
312            }
313        }
314
315        false
316    }
317
318    match ty {
319        Type::Array(ty) => type_mentions_ident(ty.elem.as_ref(), ident),
320        Type::Group(ty) => type_mentions_ident(ty.elem.as_ref(), ident),
321        Type::Paren(ty) => type_mentions_ident(ty.elem.as_ref(), ident),
322        Type::Slice(ty) => type_mentions_ident(ty.elem.as_ref(), ident),
323        Type::Ptr(ty) => type_mentions_ident(ty.elem.as_ref(), ident),
324        Type::Reference(ty) => type_mentions_ident(ty.elem.as_ref(), ident),
325        Type::Tuple(ty) => ty.elems.iter().any(|ty| type_mentions_ident(ty, ident)),
326        Type::BareFn(ty) => {
327            ty.inputs.iter().any(|arg| type_mentions_ident(&arg.ty, ident))
328                || matches!(&ty.output, ReturnType::Type(_, ty) if type_mentions_ident(ty, ident))
329        },
330        Type::ImplTrait(ty) => ty
331            .bounds
332            .iter()
333            .any(|b| matches!(b, TypeParamBound::Trait(b) if path_mentions_ident(&b.path, ident))),
334        Type::TraitObject(ty) => ty
335            .bounds
336            .iter()
337            .any(|b| matches!(b, TypeParamBound::Trait(b) if path_mentions_ident(&b.path, ident))),
338        Type::Path(ty) => {
339            (match &ty.qself {
340                Some(qself) => type_mentions_ident(qself.ty.as_ref(), ident),
341                None => false,
342            }) || path_mentions_ident(&ty.path, ident)
343        },
344        _ => false,
345    }
346}

348#[inline]
349pub(crate) fn dereference(ty: &Type) -> &Type {
350    if let Type::Reference(ty) = ty { dereference(ty.elem.as_ref()) } else { ty }
351}

353#[inline]
354pub(crate) fn dereference_changed(ty: &Type) -> (&Type, bool) {
355    if let Type::Reference(ty) = ty { (dereference(ty.elem.as_ref()), true) } else { (ty, false) }
