6/// The integer type that holds the discriminant values of an enum.
7///
8/// The enum comparison handlers order variants by their discriminant values, and this type is the one those values are compared as, so it has to match the type the compiler picks for the enum.
9pub(crate) enum DiscriminantType {
10    ISize,
11    I8,
12    I16,
13    I32,
14    I64,
15    I128,
16    USize,
17    U8,
18    U16,
19    U32,
20    U64,
21    U128,
22}

27        match s.as_ref() {
28            "i8" => Some(Self::I8),
29            "i16" => Some(Self::I16),
30            "i32" => Some(Self::I32),
31            "i64" => Some(Self::I64),
32            "i128" => Some(Self::I128),
33            "isize" => Some(Self::ISize),
34            "u8" => Some(Self::U8),
35            "u16" => Some(Self::U16),
36            "u32" => Some(Self::U32),
37            "u64" => Some(Self::U64),
38            "u128" => Some(Self::U128),
39            "usize" => Some(Self::USize),

42    }

70impl DiscriminantType {
71    /// Determines the discriminant type of an enum together with the discriminant value of every variant, in declaration order.
72    ///
73    /// An explicit `#[repr(intN)]` attribute decides the type; otherwise the smallest integer type that covers the range of the discriminant values is chosen, mirroring what the compiler does for the default representation. The values themselves are always computed by evaluating the explicit discriminant expressions (implicit ones count up from the previous value), which is exactly the order `core::mem::discriminant` would follow.
74    pub(crate) fn from_ast(ast: &DeriveInput) -> syn::Result<(Self, Vec<i128>)> {
75        if let Data::Enum(data) = &ast.data {
76            let mut repr = None;
77
78            for attr in ast.attrs.iter() {
79                if attr.path().is_ident("repr") {
80                    // #[repr(u8)], #[repr(u16)], ..., etc.
81                    if let Meta::List(list) = &attr.meta {
82                        let result =
83                            list.parse_args_with(Punctuated::<Ident, Token![,]>::parse_terminated)?;
84
85                        for value in result {
86                            if let Some(t) = Self::parse_str(value.to_string()) {
87                                repr = Some(t);
88                                break;
89                            }
90                        }
91                    }
92
93                    if repr.is_some() {
94                        break;
95                    }
96                }
97            }
98
99            // Track the smallest and largest discriminant values while walking the variants; `counter` is the value the next variant gets when it has no explicit discriminant.
100            let mut values = Vec::with_capacity(data.variants.len());
101            let mut min = i128::MAX;
102            let mut max = i128::MIN;
103            let mut counter = 0i128;
104
105            for variant in data.variants.iter() {
106                if let Some((_, exp)) = variant.discriminant.as_ref() {
107                    match exp {
108                        Expr::Lit(lit) => {
109                            if let Lit::Int(lit) = &lit.lit {
110                                counter = lit
111                                    .base10_parse()
112                                    .map_err(|error| syn::Error::new_spanned(lit, error))?;
113                            } else {
114                                return Err(syn::Error::new_spanned(lit, "not an integer"));
115                            }
116                        },
117                        Expr::Unary(unary) => {
118                            if let UnOp::Neg(_) = unary.op {
119                                if let Expr::Lit(lit) = unary.expr.as_ref() {
120                                    if let Lit::Int(lit) = &lit.lit {
121                                        match lit.base10_parse::<i128>() {
122                                            Ok(i) => {
123                                                counter = -i;
124                                            },
125                                            Err(error) => {
126                                                // overflow
127                                                if lit.base10_digits()
128                                                    == "170141183460469231731687303715884105728"
129                                                {
130                                                    counter = i128::MIN;
131                                                } else {
132                                                    return Err(syn::Error::new_spanned(
133                                                        lit, error,
134                                                    ));
135                                                }
136                                            },
137                                        }
138                                    } else {
139                                        return Err(syn::Error::new_spanned(lit, "not an integer"));
140                                    }
141                                } else {
142                                    return Err(syn::Error::new_spanned(
143                                        &unary.expr,
144                                        "not a literal",
145                                    ));
146                                }
147                            } else {
148                                return Err(syn::Error::new_spanned(
149                                    unary.op,
150                                    "this operation is not allow here",
151                                ));
152                            }
153                        },
154                        _ => return Err(syn::Error::new_spanned(exp, "not a literal")),
155                    }
156                }
157
158                values.push(counter);
159
160                if min > counter {
161                    min = counter;
162                }
163
164                if max < counter {
165                    max = counter;
166                }
167
168                counter = counter.saturating_add(1);
169            }
170
171            let discriminant_type = if let Some(t) = repr {
172                t
173            } else if min >= i8::MIN as i128 && max <= i8::MAX as i128 {
174                Self::I8
175            } else if min >= i16::MIN as i128 && max <= i16::MAX as i128 {
176                Self::I16
177            } else if min >= i32::MIN as i128 && max <= i32::MAX as i128 {
178                Self::I32
179            } else if min >= i64::MIN as i128 && max <= i64::MAX as i128 {
180                Self::I64
181            } else {
182                Self::I128
183            };
184
185            Ok((discriminant_type, values))
186        } else {
187            Err(syn::Error::new_spanned(ast, "not an enum"))
188        }
189    }
