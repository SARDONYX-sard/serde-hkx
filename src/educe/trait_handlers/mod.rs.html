53/// Shared state that flows through all trait handlers during a single `#[derive(Educe)]` expansion.
54///
55/// Its main job is to let a trait inherit the where predicates of its prerequisite traits, e.g. `Ord` inherits the predicates of `Eq` and `PartialOrd`.
56#[derive(Default)]
57pub(crate) struct TraitHandlerContext {
58    /// The final where predicates that each handled trait has actually emitted, keyed by trait.
59    #[cfg(any(
60        feature = "Clone",
61        feature = "Copy",
62        feature = "Eq",
63        feature = "Ord",
64        feature = "PartialEq",
65        feature = "PartialOrd"
66    ))]
67    final_predicates: HashMap<Trait, WherePredicates>,
68}
69
70impl TraitHandlerContext {
71    /// Records the where predicates that a trait impl has emitted, so that traits handled later can inherit them.
72    #[cfg(any(
73        feature = "Clone",
74        feature = "Copy",
75        feature = "Eq",
76        feature = "Ord",
77        feature = "PartialEq",
78        feature = "PartialOrd"
79    ))]
80    pub(crate) fn record(&mut self, t: Trait, predicates: &WherePredicates) {
81        self.final_predicates.insert(t, predicates.clone());
82    }
83
84    /// Appends the recorded predicates of every prerequisite trait to `own`, skipping predicates that are already present.
85    ///
86    /// Prerequisites that were not handled by Educe (e.g. implemented manually by the user) simply have no record and contribute nothing.
87    #[cfg(any(feature = "Copy", feature = "Eq", feature = "Ord", feature = "PartialOrd"))]
88    pub(crate) fn inherit_from(&self, prerequisites: &[Trait], own: &mut WherePredicates) {
89        // Compare predicates by their token strings because `WherePredicate` implements neither `Eq` nor `Hash`.
90        let mut seen: HashSet<String> =
91            own.iter().map(|predicate| predicate.to_token_stream().to_string()).collect();
92
93        for prerequisite in prerequisites {
94            if let Some(predicates) = self.final_predicates.get(prerequisite) {
95                for predicate in predicates {
96                    if seen.insert(predicate.to_token_stream().to_string()) {
97                        own.push(predicate.clone());
98                    }
99                }
100            }
101        }
102    }
103}
