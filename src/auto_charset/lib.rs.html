8/// Converts an arbitrary byte slice to a UTF-8 `String` using automatic encoding detection.
9///
10/// Supported encodings include UTF-8, Shift_JIS, EUC-JP, ISO-2022-JP, etc.
11/// Returns an `io::Error` if the input is empty or cannot be decoded properly.
12///
13/// # Errors
14/// If input couldn't read.
15pub fn decode_to_utf8(input: Vec<u8>) -> io::Result<String> {
16    if input.is_empty() {
17        return Ok(String::new());
18    }
19
20    // Detect encoding
21    let mut detector = EncodingDetector::new(Iso2022JpDetection::Allow);
22    detector.feed(&input, true);
23    let encoding = detector.guess(Some(b"utf-8"), Utf8Detection::Allow);
24
25    // Decode to UTF-8
26    let (decoded, _, had_errors) = encoding.decode(&input);
27
28    if had_errors {
29        return Err(io::Error::new(
30            ErrorKind::InvalidData,
31            "Failed to decode all characters",
32        ));
33    }
34
35    Ok(match decoded {
36        std::borrow::Cow::Borrowed(_s) => unsafe { String::from_utf8_unchecked(input) },
37        std::borrow::Cow::Owned(s) => s,
38    })
39}

41/// Converts an arbitrary byte slice to a UTF-8 `String` using automatic encoding detection.
42///
43/// Supported encodings include UTF-8, Shift_JIS, EUC-JP, ISO-2022-JP, etc.
44/// Returns an `io::Error` if the input is empty or cannot be decoded properly.
45///
46/// # Errors
47/// If input couldn't read.
48pub fn decode_str_to_utf8(input: &[u8]) -> io::Result<Cow<'_, str>> {
49    if input.is_empty() {
50        return Ok(Cow::Borrowed(""));
51    }
52    let mut detector = EncodingDetector::new(Iso2022JpDetection::Allow);
53    detector.feed(input, true);
54    let encoding = detector.guess(Some(b"utf-8"), Utf8Detection::Allow);
55    let (decoded, _, had_errors) = encoding.decode(input);
56    if had_errors {
57        return Err(io::Error::new(
58            ErrorKind::InvalidData,
59            "Failed to decode all characters",
60        ));
61    }
62    Ok(decoded)
63}

69    #[test]
70    fn test_utf8() {
71        let input = b"Hello, world!".to_vec();
72        let result = decode_to_utf8(input).unwrap();
73        assert_eq!(result, "Hello, world!");
74    }

76    #[test]
77    fn test_shift_jis() {
78        // "こんにちは" in Shift_JIS
79        let shift_jis_bytes = [0x82, 0xb1, 0x82, 0xf1, 0x82, 0xc9, 0x82, 0xbf, 0x82, 0xcd].to_vec();
80        let result = decode_to_utf8(shift_jis_bytes).unwrap();
81        assert_eq!(result, "こんにちは");
82    }
83
84    #[test]
85    fn test_empty_input() {
86        let result = decode_to_utf8(b"".to_vec()).unwrap();
87        assert_eq!(result, "");
88    }
89
90    #[test]
91    fn test_invalid_bytes() {
92        let bytes = [0xff, 0xfe, 0xfd].to_vec();
