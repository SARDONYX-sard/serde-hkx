3/// # Transform
4///
5/// # C++ Info
6/// - name: `hkTransform`
7/// - type_size: ` 64`(x86)/` 64`(x86_64)
8/// - align: ` 16`(x86)/` 16`(x86_64)
9///
10/// # XML representation
11/// - [`Vector4::w`] (4th) isn't used.
12/// ```xml
13///          <!--                             Matrix3 rotation                                --><!--   Vector4 transition  -->
14/// <hkparam>(0.000000 0.000000 0.000000)(0.000000 0.000000 0.000000)(0.000000 0.000000 0.000000)(-0.000000 0.000000 -0.000000)</hkparam>
15/// ```
16#[repr(C, align(16))]
17#[cfg_attr(feature = "json_schema", derive(schemars::JsonSchema))]
18#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
19#[derive(Debug, Clone, Default, PartialEq, PartialOrd)]
20pub struct Transform {
21    /// # C++ Info
22    /// - name: `rotation`(ctype: `hkRotation`)
23    /// - offset: `  0`(x86)/`  0`(x86_64)
24    /// - type_size: ` 48`(x86)/` 48`(x86_64)
25    pub rotation: Rotation,
26    /// # C++ Info
27    /// - name: `transition`(ctype: `hkVector4`)
28    /// - offset: ` 48`(x86)/` 48`(x86_64)
29    /// - type_size: ` 16`(x86)/` 16`(x86_64)
30    ///
31    /// # NOTE
32    /// - `Vector4::w`(4th) isn't used(always 0.0).
33    pub transition: Vector4,
34}
35
36const _: () = assert!(core::mem::size_of::<Transform>() == 64);

38impl core::fmt::Display for Transform {
39    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
40        write!(
41            f,
42            "{}({:.06} {:.06} {:.06})",
43            self.rotation, self.transition.x, self.transition.y, self.transition.z,
44        )
45    }
46}

48impl Transform {

58    pub fn to_le_bytes(&self) -> [u8; 64] {
59        let mut bytes = [0_u8; 64];

63    }

68        bytes[48..64].copy_from_slice(&self.transition.to_be_bytes());
